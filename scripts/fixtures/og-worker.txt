import { renderOG } from "../../apps/api/og-render";
import { HELP_VIEWS } from "../../packages/protocol/help";
import { sha256 } from "viem";
import red from "./red.png";
import blue from "./blue.png";
import yellow from "./yellow.png";
import magenta from "./magenta.png";
import large from "./large.png";
import portrait from "./portrait.png";
import landscape from "./landscape.png";
import avatar from "./avatar.png";
import cover from "./cover.png";
const media = new Map<string, Uint8Array>();
const art = [red, blue, yellow, magenta, large, portrait, landscape].map(
  (b, i) => {
    const bytes = new Uint8Array(b),
      hash = sha256(bytes);
    media.set(`art/${hash.slice(2)}.png`, bytes);
    return {
      hash,
      width: i === 4 ? 6000 : i === 6 ? 640 : 320,
      height: i === 4 ? 4000 : i === 5 ? 640 : 320,
      title: [
        "Ruby lines",
        "Blue lines",
        "Golden lines",
        "Fourth artwork",
        "24 megapixel original",
        "Portrait",
        "Landscape",
      ][i],
    };
  },
);
const address = "0x1111111111111111111111111111111111111111";
const refs = [avatar, cover].map((b) => {
  const bytes = new Uint8Array(b),
    hash = sha256(bytes);
  media.set(`profile-media/${address}/${hash.slice(2)}.png`, bytes);
  return hash;
});
const env = {
  MEDIA: {
    get: async (key: string) => {
      const b = media.get(key);
      return b ? { size: b.length, arrayBuffer: async () => b.buffer } : null;
    },
    put: async (key: string, bytes: Uint8Array) => media.set(key, bytes),
  },
};
export default {
  async fetch(req: Request) {
    const key = new URL(req.url).pathname.slice(1);
    const selected = key.startsWith("help-")
      ? []
      : key.startsWith("profile") || key === "long-title"
        ? art.slice(0, 3)
        : key === "large"
          ? [art[4]]
          : key === "portrait"
            ? [art[5]]
            : key === "landscape"
              ? [art[6]]
              : key === "missing"
                ? [{ ...art[0], hash: "0x" + "1".repeat(64) }]
                : art.slice(0, Number(key));
    const help =
      key === "help-basics"
        ? HELP_VIEWS.basics
        : key === "help-technical"
          ? HELP_VIEWS.cryptography
          : null;
    const started = Date.now();
    const image = await renderOG(
      {
        version: 1,
        deployment: "fixture",
        path: "/",
        imagePath: "",
        title: key.startsWith("profile")
          ? "Profile artwork fixture"
          : key === "long-title"
            ? "A very long collectible title with <markup> & symbols, emoji 🎨 and enough words to fill all the available space"
            : key === "4"
              ? "3 artwork fixture"
              : `${key} artwork fixture`,
        description: "",
        label: "Renderer acceptance fixture",
        subtitle: "PUBLIC TEST PIXELS",
        ...help,
        art: selected,
        ...(key.startsWith("profile")
          ? {
              profileMedia: {
                address,
                avatar: key === "profile" ? refs[0] : "0x" + "4".repeat(64),
                cover: key === "profile" ? refs[1] : "0x" + "5".repeat(64),
              },
            }
          : {}),
        revisionData: null,
      },
      env as any,
    );
    return new Response(image, {
      headers: {
        "content-type": "image/png",
        "x-render-ms": String(Date.now() - started),
      },
    });
  },
};
